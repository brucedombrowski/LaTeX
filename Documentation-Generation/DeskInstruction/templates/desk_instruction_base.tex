% Individual DIs should define variables and \input this file.

% Required Variables:
%   \DITitle          - Main title (e.g., "KuIP Payload Integration")
%   \DIID             - Document identifier (e.g., DI-PSI-319)
%   \DIDate           - Publication / revision date
%   \DIVersion        - Version string (e.g., "v1.0")
%   \DIAuthor         - Author or responsible organization
%   \DISubject        - One-line subject description
%   \DIOPR            - Office of Primary Responsibility
%
% Optional Variables:
%   \DIClassification - Classification banner (default: UNCLASSIFIED)
%   \DILogoPath       - Path to header logo (default: none)
%   \DITitlePageExtras- Additional content for title page
%
% Required Content Commands:
%   \DIContent        - Main body of the desk instruction
%
% Optional Content Commands:
%   \DIAppendix       - Appendix content (default: empty)
%   \DIReferences     - Reference list (default: empty)
%   \DIGlossary       - Acronym / glossary content (default: empty)
%
% Custom Commands Provided:
%   \distep{title}          - Begin a numbered procedure step
%   \dinote{text}           - Blue info-note callout box
%   \dicaution{text}        - Yellow caution callout box
%   \diwarning{text}        - Red warning callout box
%   \discript{text}         - Green "what to say" script box
%   \digotcha{text}         - Orange gotcha / common-mistake box
%   \diref{doc}{para}{desc} - Inline reference callout
%   \ditable{caption}{cols}{rows} - Wrapped table with caption
% ============================================================================

\newcommand{\dicalloutbox}[4]{%
  % #1=border color, #2=bg color, #3=label, #4=body
  \par\vspace{4pt}%
  \noindent\setlength{\fboxsep}{6pt}\setlength{\fboxrule}{0.6pt}%
  \fcolorbox{#1}{#2}{%
    \begin{minipage}{\dimexpr\linewidth-2\fboxsep-2\fboxrule\relax}
      {\small\bfseries\color{#1}#3}\par\vspace{2pt}
      {\small #4}
    \end{minipage}%
  }\par\vspace{4pt}%
}

\newcommand{\dinote}[1]{\dicalloutbox{diNoteBorder}{diNoteBlue}{NOTE}{#1}}
\newcommand{\dicaution}[1]{\dicalloutbox{diCautionBorder}{diCautionYellow}{CAUTION}{#1}}
